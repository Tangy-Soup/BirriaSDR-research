cmake_minimum_required(VERSION 3.16)
project(birriasdr VERSION 0.1.0 LANGUAGES C)
include(GNUInstallDirs)
include(CTest)
option(BUILD_SHARED_LIBS "Build a shared library" ON)
option(BIRRIASDR_SANITIZE "Enable address/undefined behavior sanitizers" OFF)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(-Wall -Wextra -Wpedantic -Werror)
endif()

find_package(Threads REQUIRED)
find_package(PkgConfig QUIET)
if(PkgConfig_FOUND)
    pkg_check_modules(PC_LIBUSB QUIET libusb-1.0)
endif()
find_path(LIBUSB_INCLUDE_DIR libusb.h HINTS ${PC_LIBUSB_INCLUDE_DIRS} PATH_SUFFIXES libusb-1.0)
find_library(LIBUSB_LIBRARY NAMES usb-1.0 HINTS ${PC_LIBUSB_LIBRARY_DIRS})
include(FindPackageHandleStandardArgs)
find_package_handle_standard_args(LIBUSB REQUIRED_VARS LIBUSB_LIBRARY LIBUSB_INCLUDE_DIR)

set(BIRRIASDR_SOURCES
    src/device.c src/io.c src/usb.c src/usb_stream.c src/stream.c
    src/chips/mt7921_boot.c src/chips/mt7921_rf.c src/chips/mt7921_gain.c
    src/chips/mt7921_capture.c src/chips/mt7921_frame.c src/chips/mt7925.c)
add_library(birriasdr_objects OBJECT ${BIRRIASDR_SOURCES})
set_target_properties(birriasdr_objects PROPERTIES POSITION_INDEPENDENT_CODE ON C_VISIBILITY_PRESET hidden)
target_compile_features(birriasdr_objects PUBLIC c_std_11)
target_compile_definitions(birriasdr_objects PRIVATE _POSIX_C_SOURCE=200809L)
if(APPLE)
    # Strict POSIX mode hides flock() and O_NOFOLLOW on Darwin.
    target_compile_definitions(birriasdr_objects PRIVATE _DARWIN_C_SOURCE)
endif()
target_include_directories(birriasdr_objects PUBLIC include PRIVATE src ${LIBUSB_INCLUDE_DIR})
target_link_libraries(birriasdr_objects PUBLIC Threads::Threads)
if(BIRRIASDR_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
    target_compile_options(birriasdr_objects PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
endif()
add_library(birriasdr $<TARGET_OBJECTS:birriasdr_objects>)
target_include_directories(birriasdr PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(birriasdr PRIVATE ${LIBUSB_LIBRARY} Threads::Threads)
set_target_properties(birriasdr PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 0)

add_executable(birria_rx examples/birria_rx.c)
target_link_libraries(birria_rx PRIVATE birriasdr)
target_compile_definitions(birria_rx PRIVATE _POSIX_C_SOURCE=200809L)
file(RELATIVE_PATH BIRRIA_BIN_TO_LIB "${CMAKE_INSTALL_FULL_BINDIR}" "${CMAKE_INSTALL_FULL_LIBDIR}")
if(APPLE)
    set_target_properties(birria_rx PROPERTIES INSTALL_RPATH "@loader_path/${BIRRIA_BIN_TO_LIB}")
elseif(UNIX)
    set_target_properties(birria_rx PROPERTIES INSTALL_RPATH "$ORIGIN/${BIRRIA_BIN_TO_LIB}")
endif()

if(BUILD_TESTING)
    add_executable(test_core tests/test_core.c $<TARGET_OBJECTS:birriasdr_objects>)
    target_include_directories(test_core PRIVATE include src ${LIBUSB_INCLUDE_DIR})
    target_link_libraries(test_core PRIVATE ${LIBUSB_LIBRARY} Threads::Threads)
    add_test(NAME core COMMAND test_core)
    add_executable(test_rf tests/test_rf.c $<TARGET_OBJECTS:birriasdr_objects>)
    target_include_directories(test_rf PRIVATE include src ${LIBUSB_INCLUDE_DIR})
    target_link_libraries(test_rf PRIVATE ${LIBUSB_LIBRARY} Threads::Threads)
    add_test(NAME rf_sequence COMMAND test_rf ${CMAKE_CURRENT_SOURCE_DIR}/tests/python-channel6.trace)
    add_executable(test_usb_stream tests/test_usb_stream.c $<TARGET_OBJECTS:birriasdr_objects>)
    target_include_directories(test_usb_stream PRIVATE include src ${LIBUSB_INCLUDE_DIR})
    target_link_libraries(test_usb_stream PRIVATE ${LIBUSB_LIBRARY} Threads::Threads)
    add_test(NAME usb_lifecycle COMMAND test_usb_stream)
    add_executable(test_hardware tests/test_hardware.c)
    target_link_libraries(test_hardware PRIVATE birriasdr)
    # Deliberately not registered with CTest: this resets a physical dongle.
endif()

file(RELATIVE_PATH BIRRIA_PC_TO_PREFIX "${CMAKE_INSTALL_FULL_LIBDIR}/pkgconfig" "${CMAKE_INSTALL_PREFIX}")
file(RELATIVE_PATH BIRRIA_PC_TO_INCLUDE "${CMAKE_INSTALL_FULL_LIBDIR}/pkgconfig" "${CMAKE_INSTALL_FULL_INCLUDEDIR}")
configure_file(birriasdr.pc.in birriasdr.pc @ONLY)
install(TARGETS birriasdr birria_rx
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
    ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES include/birriasdr.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/birriasdr.pc DESTINATION ${CMAKE_INSTALL_LIBDIR}/pkgconfig)
install(FILES LICENSE LICENSES/BSD-3-Clause-Clear.txt DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/birriasdr)
install(FILES NOTICE README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/birriasdr)
install(DIRECTORY docs/ DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/birriasdr)
