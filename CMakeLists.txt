cmake_minimum_required(VERSION 3.16)
project(birriasdr VERSION 0.1.0 LANGUAGES C)
include(GNUInstallDirs)
include(CTest)
option(BUILD_SHARED_LIBS "Build a shared library" ON)
option(BIRRIASDR_SANITIZE "Enable address/undefined behavior sanitizers" OFF)
option(BIRRIASDR_SOAPY "Build the SoapySDR module (needs C++17 and SoapySDR)" OFF)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(-Wall -Wextra -Wpedantic -Werror)
endif()

find_package(Threads REQUIRED)
find_package(PkgConfig QUIET)
if(PkgConfig_FOUND)
    pkg_check_modules(PC_LIBUSB QUIET libusb-1.0)
endif()
find_path(LIBUSB_INCLUDE_DIR libusb.h HINTS ${PC_LIBUSB_INCLUDE_DIRS} PATH_SUFFIXES libusb-1.0)
find_library(LIBUSB_LIBRARY NAMES usb-1.0 HINTS ${PC_LIBUSB_LIBRARY_DIRS})
include(FindPackageHandleStandardArgs)
find_package_handle_standard_args(LIBUSB REQUIRED_VARS LIBUSB_LIBRARY LIBUSB_INCLUDE_DIR)

set(BIRRIASDR_SOURCES
    src/device.c src/io.c src/usb.c src/usb_stream.c src/stream.c src/decimate.c
    src/chips/mt7921_boot.c src/chips/mt7921_rf.c src/chips/mt7921_tune.c src/chips/mt7921_gain.c
    src/chips/mt7921_capture.c src/chips/mt7921_filter.c src/chips/mt7921_detect.c src/chips/mt7921_chain.c
    src/chips/mt7921_frame.c src/chips/mt7925.c)
add_library(birriasdr_objects OBJECT ${BIRRIASDR_SOURCES})
set_target_properties(birriasdr_objects PROPERTIES POSITION_INDEPENDENT_CODE ON C_VISIBILITY_PRESET hidden)
target_compile_features(birriasdr_objects PUBLIC c_std_11)
target_compile_definitions(birriasdr_objects PRIVATE _POSIX_C_SOURCE=200809L)
if(APPLE)
    # Strict POSIX mode hides flock() and O_NOFOLLOW on Darwin.
    target_compile_definitions(birriasdr_objects PRIVATE _DARWIN_C_SOURCE)
endif()
target_include_directories(birriasdr_objects PUBLIC include PRIVATE src ${LIBUSB_INCLUDE_DIR})
target_link_libraries(birriasdr_objects PUBLIC Threads::Threads)
if(BIRRIASDR_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
    target_compile_options(birriasdr_objects PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
endif()
add_library(birriasdr $<TARGET_OBJECTS:birriasdr_objects>)
target_include_directories(birriasdr PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(birriasdr PRIVATE ${LIBUSB_LIBRARY} Threads::Threads)
set_target_properties(birriasdr PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 0)

add_executable(birria_rx examples/birria_rx.c)
target_link_libraries(birria_rx PRIVATE birriasdr)
target_compile_definitions(birria_rx PRIVATE _POSIX_C_SOURCE=200809L)
add_executable(birria_ble examples/birria_ble.c)
target_link_libraries(birria_ble PRIVATE birriasdr Threads::Threads m)
target_compile_definitions(birria_ble PRIVATE _POSIX_C_SOURCE=200809L _DEFAULT_SOURCE)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    # The FIR and discriminator loops only vectorize with reassociation.
    target_compile_options(birria_ble PRIVATE -O3 -ffast-math)
endif()
file(RELATIVE_PATH BIRRIA_BIN_TO_LIB "${CMAKE_INSTALL_FULL_BINDIR}" "${CMAKE_INSTALL_FULL_LIBDIR}")
if(APPLE)
    set_target_properties(birria_rx birria_ble PROPERTIES INSTALL_RPATH "@loader_path/${BIRRIA_BIN_TO_LIB}")
elseif(UNIX)
    set_target_properties(birria_rx birria_ble PROPERTIES INSTALL_RPATH "$ORIGIN/${BIRRIA_BIN_TO_LIB}")
endif()
